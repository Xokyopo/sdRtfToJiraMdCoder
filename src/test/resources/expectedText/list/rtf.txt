                    <li>Сервисы (Business Logic)

                        <ul>
                            <li>AuthService

                                <ul>
                                    <li>registerUser()

                                        <ul>
                                            <li>Входные данные: RegistrationRequest</li>
                                            <li>Проверка: уникальность email</li>
                                            <li>Хэширование: пароль bcrypt</li>
                                            <li>Ответ: AuthResponse с JWT</li>
                                        </ul>
                                    </li>
                                    <li>login()

                                        <ul>
                                            <li>Входные данные: LoginRequest</li>
                                            <li>Аутентификация: email/password</li>
                                            <li>Генерация: access/refresh tokens</li>
                                            <li>Ответ: AuthResponse</li>
                                        </ul>
                                    </li>
                                    <li>validateToken()

                                        <ul>
                                            <li>Входные данные: JWT token</li>
                                            <li>Парсинг: claims extraction</li>
                                            <li>Проверка: signature, expiration</li>
                                            <li>Ответ: Authentication object</li>
                                        </ul>
                                    </li>
                                    <li>refreshToken()

                                        <ul>
                                            <li>Входные данные: refresh token</li>
                                            <li>Валидация: token existence</li>
                                            <li>Генерация: new access token</li>
                                            <li>Ответ: new AuthResponse</li>
                                        </ul>
                                    </li>
                                </ul>
                            </li>
                            <li>UserService

                                <ul>
                                    <li>createUser()

                                        <ul>
                                            <li>Входные данные: UserCreateDTO</li>
                                            <li>Преобразование: DTO -&gt; Entity</li>
                                            <li>Сохранение: user repository</li>
                                            <li>Ответ: UserDTO</li>
                                        </ul>
                                    </li>
                                    <li>updateUser()

                                        <ul>
                                            <li>Входные данные: UserUpdateDTO</li>
                                            <li>Поиск: user by id</li>
                                            <li>Обновление: только разрешенные поля</li>
                                            <li>Ответ: updated UserDTO</li>
                                        </ul>
                                    </li>
                                    <li>deleteUser()

                                        <ul>
                                            <li>Входные данные: userId</li>
                                            <li>Проверка: существование пользователя</li>
                                            <li>Soft delete: mark as inactive</li>
                                            <li>Ответ: void</li>
                                        </ul>
                                    </li>
                                    <li>getUserProfile()

                                        <ul>
                                            <li>Входные данные: userId</li>
                                            <li>Загрузка: связанные данные</li>
                                            <li>Преобразование: Entity -&gt; ProfileDTO</li>
                                            <li>Ответ: UserProfileDTO</li>
                                        </ul>
                                    </li>
                                </ul>
                            </li>
                            <li>ProductService

                                <ul>
                                    <li>getAllProducts()

                                        <ul>
                                            <li>Параметры: фильтры, пагинация</li>
                                            <li>Запрос: product repository</li>
                                            <li>Преобразование: Entity -&gt; DTO</li>
                                            <li>Ответ: Page&lt;ProductDTO&gt;</li>
                                        </ul>
                                    </li>
                                    <li>getProductById()

                                        <ul>
                                            <li>Входные данные: productId</li>
                                            <li>Поиск: product repository</li>
                                            <li>Проверка: is active product</li>
                                            <li>Ответ: ProductDetailDTO</li>
                                        </ul>
                                    </li>
                                    <li>searchProducts()

                                        <ul>
                                            <li>Входные данные: search query</li>
                                            <li>Поиск: по title, description</li>
                                            <li>Ранжирование: по релевантности</li>
                                            <li>Ответ: List&lt;ProductDTO&gt;</li>
                                        </ul>
                                    </li>
                                    <li>updateStock()

                                        <ul>
                                            <li>Входные данные: productId, quantity</li>
                                            <li>Атомарность: <span class="atwho-query">@Transactional</span></li>
                                            <li>Проверка: достаточное количество</li>
                                            <li>Ответ: updated ProductDTO</li>
                                        </ul>
                                    </li>
                                </ul>
                            </li>
                            <li>OrderService

                                <ul>
                                    <li>createOrder()

                                        <ul>
                                            <li>Входные данные: OrderCreateRequest</li>
                                            <li>Проверка: доступность товаров</li>
                                            <li>Расчет: общая стоимость</li>
                                            <li>Сохранение: order + order items</li>
                                        </ul>
                                    </li>
                                    <li>calculateTotal()

                                        <ul>
                                            <li>Входные данные: список товаров</li>
                                            <li>Расчет: цена * количество</li>
                                            <li>Учет: скидки, налоги</li>
                                            <li>Ответ: total amount</li>
                                        </ul>
                                    </li>
                                    <li>processPayment()

                                        <ul>
                                            <li>Входные данные: payment details</li>
                                            <li>Интеграция: payment gateway</li>
                                            <li>Обработка: transaction</li>
                                            <li>Ответ: payment status</li>
                                        </ul>
                                    </li>
                                    <li>sendConfirmation()

                                        <ul>
                                            <li>Входные данные: order, user</li>
                                            <li>Генерация: email template</li>
                                            <li>Отправка: email service</li>
                                            <li>Логирование: sent status
                                            </li>
                                        </ul>
                                    </li>
                                </ul>
                            </li>
